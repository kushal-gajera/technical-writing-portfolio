\begin{frame}{A repeatable documentation process}
\begin{enumerate}
\item Plan: identify audience, problem and scope.
\item Draft: organise content into a clear structure.
\item Review: verify facts, language, diagrams and references.
\item Publish: export, share and preserve editable sources.
\item Maintain: update material when the system changes.
\end{enumerate}
\end{frame}

\begin{frame}{A little mathematics}
\begin{block}{Example: arithmetic mean}
\[\bar{x}=\frac{1}{n}\sum_{i=1}^{n}x_i\]

\begin{frame}{Key takeaways}
\begin{itemize}
\item Select tools based on document purpose and audience.
\item Keep source editable and final output readable.
\item Use diagrams, tables and equations when they add meaning.
\item Use version control to make changes traceable.
\end{itemize}
